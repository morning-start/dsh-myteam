/**
 * CHG-185：成本归因的「没有数字」三档判定 —— 纯函数，可单测。
 * @param {'offline'|'loaded'} loadState  sidecar 是否读到
 * @param {object|null} tokens            getTokens 返回值（null = 没读到）
 * @returns {{tone:'offline'|'no-usage'|'partial'|'ok', text:string}}
 */
function costDegradeOf(loadState, tokens) {
  if (loadState === 'offline') return { tone: 'offline', text: t('cost.offline') }
  if (!tokens) return { tone: 'no-usage', text: t('cost.noUsage') }
  const hasAny = Number(tokens.input ?? 0) + Number(tokens.output ?? 0) + Number(tokens.cached ?? 0) > 0
  if (!hasAny) return { tone: 'no-usage', text: t('cost.noUsage') }
  if (tokens.input === undefined || tokens.output === undefined) return { tone: 'partial', text: t('cost.partial') }
  return { tone: 'ok', text: '' }
}
```

degrade.tone === 'ok'

  : j('p', { className: `mt-hint mt-cost-${degrade.tone}`, children: degrade.text })
```

// plugin/src/core/members/hierarchy.ts
/**
 * CHG-187（FR-32 支柱一）：成员派生树的**纯规则层**。
 * 三件事：深度、直接子、级联顺序。都不碰 IO，方便单测与复用（宿主/面板/工具共用）。
 */
import { MTError } from '../errors.ts'

export interface MemberNode {
  id: string
  parentMemberId: string | null
  depth: number
  rootMemberId: string
}

export const depthOf = (roster: MemberNode[], id: string): number =>
  roster.find((x) => x.id === id)?.depth ?? 0

export const childrenOf = (roster: MemberNode[], id: string): MemberNode[] =>
  roster.filter((x) => x.parentMemberId === id)

/** 倒序 = 叶子先、父最后（释放级联必须这个顺序，否则父先没了子就成孤儿） */
export function cascadeOrder(roster: MemberNode[], rootId: string): string[] {
  const out: string[] = []
  const walk = (id: string) => { for (const child of childrenOf(roster, id)) walk(child.id); out.push(id) }
  walk(rootId)
  return out
}

export function assertSpawnAllowed(
  roster: MemberNode[],
  parentId: string | null,
  limits: { maxDepth: number; maxChildren: number },
): void {
  if (parentId === null) return
  const parent = roster.find((x) => x.id === parentId)
  if (!parent) throw new MTError('MT_NOT_FOUND', `parent member ${parentId} not found`)
  const depth = parent.depth + 1
  if (depth > limits.maxDepth) {
    throw new MTError('MT_DEPTH_EXCEEDED', `depth ${depth} > maxDepth ${limits.maxDepth}`)
  }
  const count = childrenOf(roster, parentId).length
  if (count >= limits.maxChildren) {
    throw new MTError('MT_CHILDREN_LIMIT', `children ${count} >= maxChildren ${limits.maxChildren}`)
  }
}
```

// plugin/src/core/index/project-board-index.ts
/**
 * CHG-188：项目级看板派生索引（`state/index/tasks.json`）。
 *
 * 三条铁律：
 *  ① **派生**：任何时刻可全量重建；删掉重算逐字段相等（t140-2）。
 *  ② **降级**：坏 JSON / 版本不符 / 团队数变了 → 回落到实时折叠，**绝不返回空看板**（t140-3）。
 *  ③ **写盘**：只走 `writeTextAtomic`（A2 门禁），落盘前带 `schema` + `builtAt`，版本不符即忽略。
 */
export const BOARD_INDEX_SCHEMA = 1

export async function buildProjectBoardIndex(projectRoot: string) {
  const live = await foldBoardLive(projectRoot)
  writeTextAtomic(boardIndexPath(projectRoot), `${JSON.stringify({ schema: BOARD_INDEX_SCHEMA, builtAt: new Date().toISOString(), ...live }, null, 2)}\n`)
  return live
}

export async function readProjectBoardIndex(projectRoot: string) {
  try {
    const raw = JSON.parse(readFileSync(boardIndexPath(projectRoot), 'utf8'))
    if (raw.schema !== BOARD_INDEX_SCHEMA) return foldBoardLive(projectRoot)
    return raw
  } catch {
    return foldBoardLive(projectRoot)   // 坏索引 → 实时折叠（诚实降级）
  }
}
```
